Массив – это структурированный тип данных, в котором число элементов фиксировано при объявлении, тип компонентов массива одинаков,
и имеется возможность прямого доступа к каждой отдельной компоненте объекта через индекс (порядковый номер).

Синтаксис:
```
тип_элемента имя[размер];
```
![](image.png)

Пример объявления и ввода элементов с клавиатура:
```
сonst int MAX_SIZE = 100;  // Максимальный размер массива
int arr[MAX_SIZE];         // Объявление статического массива
int n;                     // Реальное количество элементов

// Ввод размера
cout << "Введите количество элементов (не более " << MAX_SIZE << "): ";
cin >> n;

// Ввод элементов
cout << "Введите " << n << " элементов массива:" << endl;
for (int i = 0; i < n; i++) {
    cout << "arr[" << i << "] = ";
    cin >> arr[i];
}

// Вывод для проверки
cout << "\nВаш массив: ";
for (int i = 0; i < n; i++) {
    cout << arr[i] << " ";
}
cout << endl;
```

## Примеры заданий с массивами
### Задание 1.
В одномерном массиве, состоящем из n целых элементов вычислить сумму элементов массива, расположенных между первым и последним нулевыми элементами
```
сonst int MAX_SIZE = 100;  // Максимальный размер массива
int arr[MAX_SIZE];         // Объявление статического массива
int n; 
int firstZero = -1; //индекс первого нуля в массиве
int lastZero = 0; //индекс последнего нуля в массиве
int sum = 0;

//ввод размера
cout << "Введите количество элементов (не более " << MAX_SIZE << "): ";
cin >> n;

//ввод элементов
cout << "Введите " << n << " элементов массива:" << endl;
for (int i = 0; i < n; i++)
{
    cout << "Введите " << i + 1 << " элемент массива: ";
    cin >> arr[i]; //заполняем массив
}
for (int i = 0; i < n; i++) //поиск первого и последнего нуля
{
    if (arr[i] == 0 && firstZero == -1) {
        firstZero = i;
    }
    if (arr[i] == 0 && i > lastZero) {
        lastZero = i;
    }
}
for (int i = firstZero; i < lastZero; ++i) { //сумма элементов между первым и последним нулем
    sum += arr[i];
}
cout << "firstZero: " << firstZero << " lastZero: " << lastZero << endl;
cout << "sum = " << sum << endl;
```
